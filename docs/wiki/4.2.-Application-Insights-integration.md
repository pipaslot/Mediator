Install nuget package **Microsoft.ApplicationInsights.AspNetCore** and configure instrumentation key. Application Insights exposes Telemetry client which can be used for collecting info about action execution. You can use the following example:

```
public class TelemetryMiddleware : IMediatorMiddleware
{
    private readonly TelemetryClient _telemetryClient;

    public TelemetryMiddleware(TelemetryClient telemetryClient){
        _telemetryClient = telemetryClient
    }

    public async Task Invoke(MediatorContext context, MiddlewareDelegate next)
    {
        using (var operation = _telemetryClient.StartOperation<DependencyTelemetry>(context.ActionIdentifier))
        {
            operation.Telemetry.Type = "Mediator";
            operation.Telemetry.Success = true;
            try
            {
                await next(context);
            }
            catch (Exception e)
            {
                operation.Telemetry.Success = false;
                operation.Telemetry.Data = Serialize(context.Action);
            }
        }
    }

    private static string Serialize(object? obj)
    {
        if (obj == null)
        {
            return "NULL";
        }
        try
        {
            return JsonSerializer.Serialize(obj, _serializationOptions);
        }
        catch (Exception e)
        {
            return "Data serialization failed: " + e.Message;
        }
    }
}
```
